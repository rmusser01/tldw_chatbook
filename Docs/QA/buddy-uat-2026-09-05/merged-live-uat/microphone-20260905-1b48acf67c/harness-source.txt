    shutil.copytree(SOURCE, user_data, ignore=shutil.ignore_patterns('*.db','*.db-wal','*.db-shm','*.lock'))

from tldw_chatbook.app import TldwCli
from tldw_chatbook.Chat.console_chat_models import ConsoleMessageRole
from tldw_chatbook.Event_Handlers.TTS_Events import tts_events
from tldw_chatbook.TTS.audio_player import get_audio_player

async def main():
    from tldw_chatbook.Widgets.Console import ConsoleComposerBar
    from tldw_chatbook.Chat.console_project_instructions import ProjectInstructionControlState
    result={**IDENTITY,'microphone_authorized':True,'transcript_to_deepseek_authorized':True,'audio_saved':False,'audio_sent_to_provider':False,'transcription_provider':'faster-whisper','source_root':str(ROOT)}
    def stage(value):
        (OUT/'stage.json').write_text(json.dumps({'stage':value})+'\n')
    original=tts_events.pump
    async def observe(sink,source,**kwargs):
        outcome=await original(sink,source,**kwargs)
        result['sink']={'outcome':outcome.outcome,'bytes_fed':outcome.bytes_fed,'terminal_reason':sink.terminal_reason}
        return outcome
    tts_events.pump=observe
    app=TldwCli()
    async with app.run_test(size=(140,45)) as pilot:
        await pilot.pause(2)
        screen=app.screen
        screen.action_new_console_tab()
        await pilot.pause(.5)
        store=app.console_runtime.chat_store
        sid=store.active_session_id
        store.set_session_project_instruction_state(sid,ProjectInstructionControlState.legacy_disabled())
        composer=screen.query_one('#console-native-composer',ConsoleComposerBar)
        stage('ready')
        deadline=time.monotonic()+100
        while not (OUT/'start').exists() and time.monotonic()<deadline: await asyncio.sleep(.1)
        if not (OUT/'start').exists(): raise RuntimeError('No start signal')
        screen._request_console_dictation_start()
        deadline=time.monotonic()+40
        while screen._console_dictation_state!='recording' and time.monotonic()<deadline: await asyncio.sleep(.05)
        result['recording_started']=screen._console_dictation_state=='recording'
        if not result['recording_started']: raise RuntimeError('Microphone failed to enter recording')
        session=screen._dictation._console_dictation_session
        service=session._controller._service
        started=time.monotonic()
        stage('recording')
        states=[]
        while time.monotonic()-started<20:
            state=app.persona_buddy_controller.snapshot().state
            if not states or states[-1]!=state:states.append(state)
            await asyncio.sleep(.05)
        screen._request_console_dictation_stop()
        result['stop_requested_seconds']=round(time.monotonic()-started,2)
        stage('transcribing')
        deadline=time.monotonic()+45
        while screen._console_dictation_state!='idle' and time.monotonic()<deadline: await asyncio.sleep(.1)
        transcript=composer.draft_text
        result.update(captured_bytes=getattr(service,'captured_bytes',None),known_words_recognized={w:w in transcript.lower() for w in ('blue','notebook','ready')},transcript_characters=len(transcript),final_dictation_state=screen._console_dictation_state,dictation_session_released=screen._dictation._console_dictation_session is None,buddy_capture_states=states)
        if all(result['known_words_recognized'].values()):
            before={m.id for m in store.messages_for_session(sid)}
            result['provider_send_accepted']=await screen._send_console_message_from_visible_action()
            stage('provider')
            deadline=time.monotonic()+40
            replies=[]
            while time.monotonic()<deadline:
                run=app.console_runtime.chat_controller.run_state_for(sid)
                replies=[m for m in store.messages_for_session(sid) if m.id not in before and m.role is ConsoleMessageRole.ASSISTANT]
                if replies and run.status.value in {'completed','failed','stopped','blocked'}:break
                await asyncio.sleep(.05)
            result['provider_status']=run.status.value
            result['provider_reply_nonempty']=any(m.content.strip() for m in replies)
            await screen._sync_native_console_chat_ui()
            if result['provider_status']=='completed':
                stage('playback')
                await screen._console_read_last_response_back()
                deadline=time.monotonic()+40
                while time.monotonic()<deadline:
                    if result.get('sink',{}).get('outcome')=='drained' and app.persona_buddy_controller.snapshot().state=='idle':break
                    await asyncio.sleep(.05)
        result['final_buddy_state']=app.persona_buddy_controller.snapshot().state
    result['app_exception']=type(app._exception).__name__ if app._exception else None
    result['normal_config_unchanged']=hashlib.sha256(NORMAL.read_bytes()).hexdigest()==normal_digest
    (OUT/'microphone.json').write_text(json.dumps(result,indent=2)+'\n')
    stage('finished')
asyncio.run(main())
