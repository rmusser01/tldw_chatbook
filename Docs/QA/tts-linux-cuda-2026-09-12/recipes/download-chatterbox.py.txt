import os,json,hashlib
from pathlib import Path
root=Path('/home/ml-user/tts-linux-cuda-20260912')
os.environ['HF_HOME']=str(root/'cache/huggingface')
os.environ['HF_HUB_DISABLE_XET']='1'
from huggingface_hub import HfApi,snapshot_download
repo='ResembleAI/chatterbox'
info=HfApi().model_info(repo,files_metadata=True)
files=['ve.safetensors','t3_cfg.safetensors','s3gen.safetensors','tokenizer.json','conds.pt']
(root/'logs/chatterbox-model-revision.json').write_text(json.dumps({'repo':repo,'revision':info.sha},indent=2)+'\n')
p=snapshot_download(repo,revision=info.sha,allow_patterns=files,max_workers=3)
# Upstream from_pretrained requests main; bind its offline cache ref to the exact inspected snapshot.
(Path(p).parent.parent/'refs').mkdir(exist_ok=True)
(Path(p).parent.parent/'refs/main').write_text(info.sha)
manifest=[]
for name in files:
 path=Path(p)/name
 with path.open('rb') as f: digest=hashlib.file_digest(f,'sha256').hexdigest()
 sibling=next(s for s in info.siblings if s.rfilename==name)
 if sibling.lfs:assert sibling.lfs.sha256==digest
 manifest.append({'name':name,'path':str(path),'sha256':digest,'bytes':path.stat().st_size})
 print(name,digest,flush=True)
(root/'logs/chatterbox-assets.json').write_text(json.dumps({'repo':repo,'revision':info.sha,'files':manifest},indent=2)+'\n')
