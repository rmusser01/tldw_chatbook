"""No inference: mount the real catalog/pane and capture a reference request."""

import asyncio
import builtins
import json
import tempfile
from pathlib import Path

from chatterbox_observation import select_lab_reference
from qualify_chatterbox import setup_environment


def main():
    original_import = builtins.__import__

    def deny_models(name, *args, **kwargs):
        if name.split(".")[0] in {"torch", "chatterbox"}:
            raise AssertionError("UI probe must not import a model runtime")
        return original_import(name, *args, **kwargs)

    builtins.__import__ = deny_models
    with tempfile.TemporaryDirectory(prefix="chatterbox-reference-ui-") as directory:
        root = Path(directory)
        setup_environment(root)
        (root / "profile/config.toml").write_text(
            '[general]\nusers_name="qualification"\n[paths]\ndata_dir="'
            + str(root / "data")
            + '"\n'
        )
        asyncio.run(probe(root))


async def probe(root):
    from textual import on
    from textual.app import App
    from textual.widgets import Button, Select, TextArea

    from tldw_chatbook.Event_Handlers.STTS_Events.stts_events import (
        STTSPlaygroundGenerateEvent,
    )
    from tldw_chatbook.TTS.adapter_bootstrap import build_default_tts_service
    from tldw_chatbook.TTS.studio_preferences import StudioTTSPreferencesSnapshot
    from tldw_chatbook.TTS.TTS_Generation import (
        bind_tts_service,
        reset_tts_service_binding,
    )
    from tldw_chatbook.UI.Speech.speech_playground_pane import SpeechPlaygroundPane
    from tldw_chatbook.UI.STTS_Window import _seed_axis_defaults

    config = {
        "default_provider": "chatterbox",
        "default_model_mode": "exact",
        "default_model": "chatterbox",
        "default_voice_mode": "exact",
        "default_voice": "default",
        "default_format": "wav",
        "default_speed": 1.0,
    }
    service = build_default_tts_service(
        {"APP_TTS_CONFIG": config, "COMPREHENSIVE_CONFIG_RAW": {"app_tts": config}}
    )
    bind_tts_service(service)

    class Host(App):
        def __init__(self):
            super().__init__()
            studio = StudioTTSPreferencesSnapshot()
            global_preferences = service.preferences_snapshot()
            self.pane = SpeechPlaygroundPane(
                provider="chatterbox",
                studio_preferences=studio,
                global_preferences=global_preferences,
                axis_defaults=_seed_axis_defaults(studio, global_preferences),
            )
            self.requests = []

        def compose(self):
            yield self.pane

        @on(STTSPlaygroundGenerateEvent)
        def capture(self, event):
            self.requests.append(event.request)

    app = Host()
    try:
        async with app.run_test(size=(150, 65)) as pilot:
            for _ in range(100):
                await pilot.pause(0.02)
                if (
                    app.pane.is_mounted
                    and not app.pane.query_one("#tts-generate-btn", Button).disabled
                ):
                    break
            voices = [
                value
                for _, value in app.pane.query_one("#tts-voice-select", Select)._options
                if isinstance(value, str)
            ]
            assert app.pane.query_one("#tts-voice-select", Select).value == "default"
            path = root / "synthetic.wav"
            path.write_bytes(b"No audio is decoded by this UI admission-only probe")
            select_lab_reference(app.pane, path)
            app.pane.query_one(
                "#tts-text-input", TextArea
            ).text = "Synthetic reference request probe."
            await pilot.pause()
            app.pane.query_one("#tts-generate-btn", Button).press()
            for _ in range(100):
                await pilot.pause(0.02)
                if app.requests:
                    break
            assert len(app.requests) == 1
            assert app.requests[0].voice_id == f"custom:{path}"
            print(
                json.dumps(
                    {
                        "status": "reference_ui_admission_passed",
                        "catalog_voice_choices": voices,
                        "request_voice_kind": "custom:<selected_path>",
                        "model_runtime_imported": False,
                        "generation_executed": False,
                    }
                )
            )
    finally:
        await service.close()
        await service.wait_closed()
        reset_tts_service_binding(expected=service)


if __name__ == "__main__":
    main()
