_PROFILE_WORKER = "run_preparation_read.<locals>.invoke"


def install():
    """Return cleanup; register only original code already loaded or called later."""
    monitoring = sys.monitoring
    events = monitoring.events
    origin = time.monotonic()
    stream = sys.stderr
    labels = {}
    seen = set()
    closed = False
    closing = False
    emitted = 0
    limit = 512
    slot = None
    gate_var = ContextVar("task62_preplan_diagnostic_gate", default=None)
    local = threading.local()
    gates, profiles = [], []
    selectors = {
        ("/tldw_chatbook/" + module.replace(".", "/") + ".py", name)
        for module, names in _TARGETS.items()
        for name in names
    }

    def emit(name, *, force=False):
        nonlocal emitted
        if emitted >= limit and not force:
            return
        emitted += 1
        try:
            stream.write(
                f"PREP_DIAG elapsed={time.monotonic() - origin:.9f} "
                f"thread_cpu={time.thread_time():.9f} name={name}\n"
            )
        except BaseException:
            pass  # Diagnostic output cannot change the original operation.

    def emit_error(phase, exc):
        kind = type(exc)
        name = kind.__name__ if kind in (
            ValueError, TypeError, RuntimeError, AttributeError, KeyError,
            OSError, MemoryError, OverflowError, SystemError,
        ) else "Other"
        emit("error." + phase + "." + name)

    def stop_profile(state):
        if state["depth"]:
            if state["profile"] is not None:
                state["profile"].disable()
            state["wall"] += time.monotonic() - state["start_wall"]
            state["cpu"] += time.thread_time() - state["start_cpu"]
            state["depth"] = 0

    def enable_worker_profile(state):
        # cProfile owns one interpreter-wide slot on Python 3.12. Only the
        # finite worker gets cProfile; owner slices retain wall/CPU timing.
        state["profile"].enable()
        profiler_slot = monitoring.PROFILER_ID
        # C_RETURN/C_RAISE accompany CALL but are absent from get_events().
        # Filter all nine cProfile callbacks so other threads cannot pop its
        # original worker stack while that worker releases the GIL.
        for event in (
            events.PY_START, events.PY_RESUME, events.PY_THROW,
            events.PY_RETURN, events.PY_YIELD, events.PY_UNWIND,
            events.CALL, events.C_RETURN, events.C_RAISE,
        ):
            original = monitoring.register_callback(profiler_slot, event, None)
            if original is None:
                continue

            def thread_only(*args, original=original):
                if threading.get_ident() == state["thread"]:
                    return original(*args)

            monitoring.register_callback(profiler_slot, event, thread_only)

    def profile_event(name, event):
        gate = gate_var.get()
        if name == "_run_received_intent_bound" and event == "start" and not closing:
            if gate is not None or len(gates) >= 2:
                emit("profile.error.received_gate_overlap_or_limit")
                return
            gate = {"number": len(gates) + 1, "open": True}
            gate["token"] = gate_var.set(gate)
            gates.append(gate)
            emit(f"profile.gate{gate['number']}.start")
        if gate is None:
            return
        if name == "ConsoleChatController._accept_durable_turn" and event == "start":
            gate["open"] = False
            for state in profiles:
                if state["gate"] is gate and state["depth"]:
                    if state["thread"] == threading.get_ident():
                        stop_profile(state)  # Stop now, not at a later outer yield.
                    else:
                        emit("profile.error.worker_still_active_at_accept")
            emit(f"profile.gate{gate['number']}.accept")
            return
        if name not in _PROFILE_OWNERS and name != _PROFILE_WORKER:
            return
        kind = "worker" if name == _PROFILE_WORKER else "owner"
        states = getattr(local, "states", None)
        if states is None:
            states = local.states = {}
        key = (gate["number"], kind)
        state = states.get(key)
        if event in {"start", "resume"} and gate["open"] and not closing:
            if state is None:
                if sum(row["gate"] is gate for row in profiles) >= 4:
                    emit("profile.error.thread_limit")
                    return
                state = {
                    "gate": gate, "kind": kind, "thread": threading.get_ident(),
                    "number": len(profiles) + 1,
                    "profile": cProfile.Profile() if kind == "worker" else None,
                    "depth": 0, "wall": 0.0, "cpu": 0.0, "segments": 0,
                }
                states[key] = state
                profiles.append(state)
            if state["depth"] == 0:
                if state["profile"] is not None and (
                    sys.getprofile() is not None
                    or monitoring.get_tool(monitoring.PROFILER_ID) is not None
                ):
                    emit("profile.error.existing_or_overlapping_profiler")
                    return
                state["start_wall"], state["start_cpu"] = time.monotonic(), time.thread_time()
                state["segments"] += 1
                if state["profile"] is not None:
                    try:
                        enable_worker_profile(state)
                    except BaseException:
                        state["profile"].disable()
                        raise
            state["depth"] += 1
        elif event in {"yield", "return", "unwind"} and state is not None and state["depth"]:
            if state["depth"] == 1:
                stop_profile(state)
            else:
                state["depth"] -= 1
        if name == "_run_received_intent_bound" and event in {"return", "unwind"}:
            gate["open"] = False
            gate_var.reset(gate["token"])

    def report_profiles():
        for state in profiles:
            prefix = f"profile.gate{state['gate']['number']}.{state['kind']}{state['number']}"
            emit(f"{prefix}.segments={state['segments']}.wall={state['wall']:.9f}.cpu={state['cpu']:.9f}")
            if state["profile"] is None:
                continue
            rows = []
            for row in state["profile"].getstats():
                code = row.code
                if type(code) is not CodeType or code.co_filename == __file__:
                    continue
                filename = code.co_filename.replace("\\", "/")
                if "/tldw_chatbook/" not in filename:
                    continue
                module = filename.split("/tldw_chatbook/", 1)[1].removesuffix(".py").replace("/", ".")
                rows.append((row, module + "." + code.co_qualname))
            chosen = []
            for metric, maximum in (("totaltime", 10), ("inlinetime", 5)):
                for row, label in sorted(rows, key=lambda item: getattr(item[0], metric), reverse=True)[:maximum]:
                    if row.code in chosen:
                        continue
                    chosen.append(row.code)
                    emit(f"{prefix}.{label}.calls={row.callcount}.recursive={row.reccallcount}.total={row.totaltime:.9f}.self={row.inlinetime:.9f}")

    for candidate in (3, 4, 5):
        if monitoring.get_tool(candidate) is not None:
            continue
        try:
            monitoring.use_tool_id(candidate, "task62-original-preparation")
        except ValueError:
            continue
        slot = candidate
        break
    if slot is None:
        emit("error.tool_id_unavailable")
        return lambda: None

    def close():
        nonlocal closed, closing
        if closed:
            return
        closing = True
        for gate in gates:
            gate["open"] = False
        for state in profiles:
            if state["thread"] == threading.get_ident():
                stop_profile(state)
        if any(state["depth"] for state in profiles):
            emit("profile.error.cleanup_waiting_for_original_scope_exit")
            return  # Keep exit callbacks until original workers disable themselves.
        report_profiles()
        closed = True
        monitoring.set_events(slot, 0)
        for code, _name in tuple(labels.values()):
            monitoring.set_local_events(slot, code, 0)
        for event in (
            events.PY_START, events.PY_RESUME, events.PY_YIELD,
            events.PY_RETURN, events.PY_UNWIND, events.CALL,
        ):
            monitoring.register_callback(slot, event, None)
        monitoring.free_tool_id(slot)
        emit("closed.limit_reached" if emitted >= limit else "closed", force=True)

    def register_code(code):
        if id(code) in seen:
            return
        seen.add(id(code))
        filename = code.co_filename.replace("\\", "/")
        name = code.co_qualname
        if any(filename.endswith(suffix) and name == selected
               for suffix, selected in selectors):
            labels[id(code)] = (code, name)
            mask = events.PY_START | events.PY_RESUME | events.PY_YIELD | events.PY_RETURN
            if name in _DISCOVERY:
                mask |= events.CALL
            monitoring.set_local_events(slot, code, mask)
        for value in code.co_consts:
            if type(value) is CodeType:
                register_code(value)

    def register_callable(value):
        selected = None
        if type(value) in (staticmethod, classmethod):
            value = value.__func__
        if type(value) is MethodType:
            value = value.__func__
        for _ in range(4):
            if type(value) is not FunctionType:
                break
            register_code(value.__code__)
            label = labels.get(id(value.__code__))
            if label is not None:
                selected = label[1]
            value = value.__dict__.get("__wrapped__")
        return selected

    def scan_loaded():
        for module_name, names in _TARGETS.items():
            module = sys.modules.get("tldw_chatbook." + module_name)
            if type(module) is not ModuleType:
                continue
            for name in names:
                value = module
                for part in name.split(".<locals>.", 1)[0].split("."):
                    if type(value) not in (ModuleType, type):
                        value = None
                        break
                    value = vars(value).get(part)
                register_callable(value)

    def on_call(code, _offset, callable_object, arg0):
        if closed:
            return
        try:
            called = register_callable(callable_object)
            caller = labels.get(id(code))
            if caller is None:
                return
            if caller[1] == "ConsoleRuntime.accept_received_intent":
                # Its lazy import has completed before the original accept call
                # can create the received Task. Register, without importing.
                scan_loaded()
            if caller[1] in _CHAIN_CALLERS and called is not None:
                emit("call." + caller[1] + "->" + called)
            if (caller[1] == "_sensitive_db_paths"
                    and called == "_sensitive_db_paths.<locals>.collect"):
                # The sole permitted argument observation is whether the exact
                # nested qualification function was supplied. No value escapes.
                selected = (
                    type(arg0) is FunctionType
                    and any(arg0.__code__ is nested
                            and nested.co_qualname == "_sensitive_db_paths.<locals>.require_current"
                            for nested in code.co_consts if type(nested) is CodeType)
                )
                route = "selected" if selected else (
                    "ungrouped" if arg0 is monitoring.MISSING else "unexpected"
                )
                emit("route._sensitive_db_paths.collect." + route)
        except BaseException as exc:
            emit_error("registration", exc)
            close()

    def callback(event_name):
        def observe(code, *event_args):
            if closed:
                return
            try:
                label = labels.get(id(code))
                if label is None:
                    return
                profile_event(label[1], event_name)
                if closing:
                    if not any(state["depth"] for state in profiles):
                        close()
                    return
                if label[1] == "_sensitive_db_paths.<locals>.current":
                    if event_name == "return":
                        result = event_args[-1]
                        outcome = "true" if result is True else (
                            "false" if result is False else "unexpected"
                        )
                        emit("return." + label[1] + "." + outcome)
                elif label[1] not in _DISCOVERY_ONLY:
                    emit(event_name + "." + label[1])
                if event_name == "start" and label[1] in _RESCAN_STARTS:
                    # Only observe modules the original path has now loaded.
                    # No import, factory, or user callback is executed here.
                    scan_loaded()
                if (
                    event_name == "return"
                    and label[1] == "ConsoleExecutionBackend.start"
                ):
                    scan_loaded()
                    emit(f"registered.after_composition.{len(labels)}")
            except BaseException as exc:
                emit_error("observation", exc)
                close()
        return observe

    phase = "installation.callbacks"
    try:
        monitoring.register_callback(slot, events.CALL, on_call)
        for event, name in (
            (events.PY_START, "start"), (events.PY_RESUME, "resume"),
            (events.PY_YIELD, "yield"), (events.PY_RETURN, "return"),
            (events.PY_UNWIND, "unwind"),
        ):
            monitoring.register_callback(slot, event, callback(name))
        phase = "installation.scan"
        scan_loaded()
        phase = "installation.unwind"
        # PY_UNWIND cannot be enabled locally in Python 3.12+. Its callback
        # immediately rejects every code object absent from the exact map.
        monitoring.set_events(slot, events.PY_UNWIND)
        emit(f"installed.{len(labels)}")
    except BaseException as exc:
        emit_error(phase, exc)
        close()
    return close
