from common import *
import ast,difflib
R='/Users/macbook-dev/Documents/GitHub/tldw_chatbook/.venv/bin/ruff'
pre=json.loads((S/'task-13-preflight-selection.json').read_text());paths=[r['path'] for r in pre['private_source_map'] if r['path'].endswith('.py')]+['Tests/Architecture/test_module_size_ratchet.py']
assert run('fatal-ruff',[R,'check','--select','E9,F63,F7,F82',*paths])==0
formatted=[r['path'] for r in pre['private_format_map'] if r['versions']['proposal']['format_unchanged']]
# Include the incoming tests and the only changed ratchet file; inherited test formatting is checked separately below.
check=formatted+['Tests/Architecture/test_module_size_ratchet.py']
assert run('changed-format',[R,'format','--check',*check])==0
assert run('source-whitespace',['git','-c','gc.auto=0','diff','--check',BASE,'--',*paths])==0
rows={}
for p in paths:
 versions={}
 for label,ref in [('reviewed',BASE),('incoming',NEW),('final',None)]:
  if ref and subprocess.run(['git','-c','gc.auto=0','cat-file','-e',ref+':'+p],cwd=W,capture_output=True).returncode:continue
  raw=blob(ref,p) if ref else (W/p).read_bytes()
  argv=[R,'format','--stdin-filename',str(W/p),'-'];r=subprocess.run(argv,input=raw,cwd=W,capture_output=True);assert r.returncode==0
  a=raw.decode().splitlines(keepends=True);b=r.stdout.decode().splitlines(keepends=True);ops=[{'old':''.join(a[i:j]),'new':''.join(b[k:l])} for tag,i,j,k,l in difflib.SequenceMatcher(a=a,b=b,autojunk=False).get_opcodes() if tag!='equal']
  versions[label]={'sha256':sha(raw),'formatted_sha256':sha(r.stdout),'lines':len(a),'formatted_lines':len(b),'ast_identical':ast.dump(ast.parse(raw),include_attributes=False)==ast.dump(ast.parse(r.stdout),include_attributes=False),'edits':ops,'argv':argv}
  assert versions[label]['ast_identical']
 prior=versions.get('reviewed',{}).get('edits',[])+versions.get('incoming',{}).get('edits',[])
 unknown=[op for op in versions['final']['edits'] if op not in prior]
 rows[p]={'versions':versions,'unattributed_final_edits':unknown}
 assert not unknown,(p,unknown)
dump('formatter-inheritance.json',rows)
print('All remaining full-file formatter edits exactly attributed to reviewed or incoming source; no whole-owner edits applied.')
