import asyncio
import json
import threading
from pathlib import Path
import pytest
from textual.widgets import TextArea
from Tests.private_profile import private_profile_test
from Tests.UI.test_chunking_lab_screen import settle_lab
from Tests.UI.app_factory import _build_test_app
from tldw_chatbook.UI.Navigation.screen_registry import resolve_screen_route
from tldw_chatbook.UI.Screens import chunking_lab_screen as module

@pytest.mark.asyncio
@pytest.mark.parametrize("later",[False,True])
@private_profile_test
async def test_pending_sample_file_keeps_later_accepted_keys(request,tmp_path,monkeypatch,later):
    lab_app=_build_test_app();lab_app._initial_screen_pushed=True
    source=tmp_path/"selected.txt";source.write_text("selected file content",encoding="utf-8")
    entered,release=threading.Event(),threading.Event()
    original=module.read_sample_file
    def held(*args,**kwargs):
        result=original(*args,**kwargs)
        entered.set()
        assert release.wait(20),"audit gate not released"
        return result
    monkeypatch.setattr(module,"read_sample_file",held)
    async with lab_app.run_test(size=(120,40)) as pilot:
        screen=resolve_screen_route("chunking_lab").load_screen_class()(lab_app)
        await lab_app.push_screen(screen);await screen.wait_until_ready();await settle_lab(lab_app,screen,pilot)
        editor=screen.query_one("#lab-sample-text",TextArea)
        editor.load_text("seed");await pilot.pause();await screen.drain_edits()
        task=asyncio.create_task(screen.file_operation("sample-file",{"path":str(source)}))
        try:
            async with asyncio.timeout(20):
                while not entered.is_set():await asyncio.sleep(.01)
            if later:
                editor.focus();await pilot.pause();await pilot.press("x");await pilot.pause();await screen.drain_edits()
                assert editor.text=="xseed",("key not accepted",editor.text)
                session=screen.coordinator.session
                assert session.samples[session.view["sample_hash"]]["text"]=="xseed"
            before=editor.text;release.set();await asyncio.wait_for(task,20);await settle_lab(lab_app,screen,pilot)
            session=screen.coordinator.session;canonical=session.samples[session.view["sample_hash"]]["text"]
            record={"later":later,"before_release":before,"visible":editor.text,"canonical":canonical,"source":session.samples[session.view["sample_hash"]].get("source")}
            (tmp_path/"witness.json").write_text(json.dumps(record,indent=2),encoding="utf-8");print("WITNESS",json.dumps(record))
            expected="xseed" if later else "selected file content"
            assert editor.text==expected,record
            assert canonical==expected,record
        finally:
            release.set()
            if not task.done():await task
