def phase(child, expected):
    assert select.select([child.stdout], [], [], 15)[0], "probe child phase timeout"
    line = child.stdout.readline()
    assert line, f"child exited {child.poll()}: {child.communicate(timeout=5)[1]}"
    assert json.loads(line) == {"phase": expected}, child.communicate(timeout=5)[1]


def send(child, command):
    child.stdin.write(command + "\n")
    child.stdin.flush()


def finish(child):
    if child.poll() is None:
        child.kill()  # Failed probe containment, never application policy.
        child.wait(timeout=5)
    for stream in (child.stdin, child.stdout, child.stderr):
        stream.close()


def launch(path, policy, transaction, count):
    subprocess.run(
        [sys.executable, "-c", SEED, str(path)],
        capture_output=True,
        text=True,
        check=True,
        timeout=20,
    )
    return subprocess.Popen(
        [sys.executable, "-c", CHILD, str(path), policy, transaction, str(count)],
        stdin=subprocess.PIPE,
        stdout=subprocess.PIPE,
        stderr=subprocess.PIPE,
        text=True,
    )


def contender(path):
    result = subprocess.run(
        [
            sys.executable,
            "-I",
            "-c",
            r"""
import sqlite3, sys
c = sqlite3.connect(sys.argv[1], isolation_level=None, timeout=0.05)
try:
    c.execute("BEGIN IMMEDIATE")
except sqlite3.OperationalError as error:
    print(error.sqlite_errorcode)
else:
    print(0)
    c.rollback()
finally:
    c.close()
""",
            str(path),
        ],
        capture_output=True,
        text=True,
        check=True,
        timeout=5,
    )
    code = int(result.stdout.strip())
    assert code in (0, sqlite3.SQLITE_BUSY)
    return code


@pytest.mark.parametrize("policy", ["default", "native"])
@pytest.mark.parametrize("transaction", ["idle", "read", "write"])
@pytest.mark.parametrize("count", [1, 2])
@pytest.mark.parametrize("exit_kind", ["orderly", "abrupt"])
def test_pristine_helper_first_foreign_preservation(
    tmp_path, policy, transaction, count, exit_kind
):
    path = tmp_path / "profiles.sqlite3"
    child = launch(path, policy, transaction, count)
    observers = []
    try:
        phase(child, "ready")
        assert contender(path) == (sqlite3.SQLITE_BUSY if transaction == "write" else 0)
        for suffix in ("-wal", "-shm"):
            selected = Path(str(path) + suffix)
            selected.rename(tmp_path / ("owned-original" + suffix))
            payload = b"foreign sentinel" + suffix.encode()
            with selected.open("xb") as stream:
                stream.write(payload)
            selected.chmod(0o600)
            fd = os.open(selected, os.O_RDONLY)
            observers.append((selected, fd, os.fstat(fd), payload))
        send(child, "substituted")
        phase(child, "retained")
        with pytest.raises(ProfileRepositoryError):
            ProfileStoreLease(
                path, ProfileStoreLockMode.EXCLUSIVE, timeout_seconds=0.1
            ).acquire()
        for selected, fd, identity, payload in observers:
            assert selected.stat().st_ino == identity.st_ino
            assert os.fstat(fd).st_nlink == 1
            assert os.pread(fd, len(payload) + 1, 0) == payload
        if exit_kind == "orderly":
            send(child, "exit")
            assert child.wait(timeout=15) == 0, child.stderr.read()
            assert json.loads(child.stdout.read()) == {
                "phase": "atexit",
                "connections": count,
                "wal_exists": True,
                "shm_exists": True,
            }
        else:
            child.kill()
            assert child.wait(timeout=5) == -signal.SIGKILL
        exclusive = ProfileStoreLease(
            path, ProfileStoreLockMode.EXCLUSIVE, timeout_seconds=0.1
        ).acquire()
        exclusive.release()
        observed = {
            selected.name: {
                "named_exists": selected.exists(),
                "same_inode": selected.exists()
                and selected.stat().st_ino == identity.st_ino,
                "nlink": os.fstat(fd).st_nlink,
                "bytes_unchanged": os.pread(fd, len(payload) + 1, 0) == payload,
            }
            for selected, fd, identity, payload in observers
        }
        assert all(
            item["named_exists"]
            and item["same_inode"]
            and item["nlink"] == 1
            and item["bytes_unchanged"]
            for item in observed.values()
        ), observed
    finally:
        finish(child)
        for _, fd, _, _ in observers:
            os.close(fd)


@pytest.mark.parametrize("policy", ["default", "native"])
@pytest.mark.parametrize("transaction", ["idle", "read", "write"])
@pytest.mark.parametrize("checkpoint", [False, True])
def test_healthy_close_and_committed_data_recovery(
    tmp_path, policy, transaction, checkpoint
):
    path = tmp_path / "profiles.sqlite3"
    child = launch(path, policy, transaction, 2)
    try:
        phase(child, "ready")
        send(child, "healthy_checkpoint" if checkpoint else "healthy_close")
        phase(child, "closed")
        assert child.wait(timeout=15) == 0, child.stderr.read()
        assert json.loads(child.stdout.read())["phase"] == "atexit"
        assert Path(str(path) + "-wal").exists() == (policy == "native")
        if policy == "native" and checkpoint:
            assert Path(str(path) + "-wal").stat().st_size == 0
        exclusive = ProfileStoreLease(
            path, ProfileStoreLockMode.EXCLUSIVE, timeout_seconds=0.1
        ).acquire()
        exclusive.release()
        connection = sqlite3.connect(path)
        try:
            assert connection.execute(
                "SELECT speed FROM tts_generation_profiles"
            ).fetchall() == [(2.0,)]
            assert connection.execute("PRAGMA integrity_check").fetchone() == ("ok",)
        finally:
            connection.close()
    finally:
        finish(child)
