from tldw_chatbook.Event_Handlers.TTS_Events import tts_events
from tldw_chatbook.TTS.audio_player import get_audio_player

async def speech():
    result={**IDENTITY,'import_root':str(Path(sys.modules['tldw_chatbook'].__file__).parent), 'provider':'kokoro','microphone_opened':False}
    original=tts_events.pump
    async def observe(sink,source,**kwargs):
        outcome=await original(sink,source,**kwargs)
        result['sink']={'outcome':outcome.outcome,'bytes_fed':outcome.bytes_fed,'terminal_reason':sink.terminal_reason}
        result.setdefault('playbacks',[]).append(dict(result['sink']))
        return outcome
    tts_events.pump=observe
    original_transition=tts_events.TTSPlaybackLifecycle._transition
    def observe_transition(owner,state):
        accepted=original_transition(owner,state)
        result.setdefault('lifecycle_events',[]).append({'request_id':owner.request_id,'state':state,'accepted':accepted})
        return accepted
    tts_events.TTSPlaybackLifecycle._transition=observe_transition
    app=TldwCli()
    states=[]
    async with app.run_test(size=(140,45)) as pilot:
        await pilot.pause(2)
        screen=app.screen
        screen.action_new_console_tab()
        await pilot.pause(.5)
        await app.reconcile_persona_buddy_view()
        buddy=app.persona_buddy_controller
        assert buddy is not None
        result['visual_before']=buddy.snapshot().visual.available if buddy.snapshot().visual else False
        result['visual_reason']=buddy.snapshot().visual.reason if buddy.snapshot().visual else 'not_resolved'
        store=app.console_runtime.chat_store
        message=store.append_message(store.active_session_id,role=ConsoleMessageRole.ASSISTANT,content='Migu voice check. The blue notebook is ready.')
        await screen._sync_native_console_chat_ui()
        states.append(buddy.snapshot().state)
        await screen._console_read_last_response_back()
        started=time.monotonic()
        while time.monotonic()-started < 40:
            state=buddy.snapshot().state
            if states[-1] != state: states.append(state)
            if result.get('sink',{}).get('outcome')=='drained' and state=='idle': break
            await asyncio.sleep(.05)
        result.update(buddy_states=states,elapsed_seconds=round(time.monotonic()-started,2),speaking_cleared=screen._console_speaking_message_id is None,app_exception=type(app._exception).__name__ if app._exception else None)
        terminals_before=sum(e['accepted'] and e['state'] in {'stopped','failed'} for e in result.get('lifecycle_events',[]))
        store.append_message(store.active_session_id,role=ConsoleMessageRole.ASSISTANT,content='This is a synthetic voice ownership check. ' * 8)
        await screen._sync_native_console_chat_ui()
        await screen._console_read_last_response_back()
        ownership_start=time.monotonic()
        while time.monotonic()-ownership_start < 30 and buddy.snapshot().state != 'speaking':
            await asyncio.sleep(.03)
        result['ownership_playback_started']=buddy.snapshot().state == 'speaking'
        survivor=buddy.acquire_state(source='voice',owner='uat-independent-owner',state='speaking')
        screen.action_new_console_tab()
        await pilot.pause(.1)
        while time.monotonic()-ownership_start < 45 and sum(e['accepted'] and e['state'] in {'stopped','failed'} for e in result.get('lifecycle_events',[])) <= terminals_before:
            await asyncio.sleep(.05)
        result['terminal_ack_after_context_switch']=sum(e['accepted'] and e['state'] in {'stopped','failed'} for e in result.get('lifecycle_events',[])) > terminals_before
        result['state_after_context_switch_terminal']=buddy.snapshot().state
        result['other_voice_owner_preserved']=buddy.release_state(token=survivor)
        await pilot.pause(.2)
        result['state_after_other_owner_release']=buddy.snapshot().state
        result['speaking_presentation_cleared_after_context_switch']=screen._console_speaking_message_id is None
        await asyncio.to_thread(get_audio_player().stop)
    result['app_exception']=type(app._exception).__name__ if app._exception else None
    result['normal_config_unchanged']=hashlib.sha256(NORMAL.read_bytes()).hexdigest()==normal_digest
    (OUT/'kokoro.json').write_text(json.dumps(result,indent=2)+'\n')
    print(json.dumps(result),flush=True)

async def provider():
    from tldw_chatbook.Widgets.Console import ConsoleComposerBar
    from tldw_chatbook.Chat.console_project_instructions import ProjectInstructionControlState
    app=TldwCli()
    result={**IDENTITY,'provider':'deepseek','microphone_opened':False,'fixture':'Reply exactly: Migu live check passed.','import_root':str(Path(sys.modules['tldw_chatbook'].__file__).parent)}
    states=[]
    async with app.run_test(size=(140,45)) as pilot:
        await pilot.pause(2)
        screen=app.screen
        screen.action_new_console_tab()
        await pilot.pause(.5)
        store=app.console_runtime.chat_store
        sid=store.active_session_id
        store.set_session_project_instruction_state(sid,ProjectInstructionControlState.legacy_disabled())
        composer=screen.query_one('#console-native-composer',ConsoleComposerBar)
        composer.load_draft(result['fixture'])
        await screen._send_console_message_from_visible_action()
        start=time.monotonic()
        while time.monotonic()-start < 45:
            run=app.console_runtime.chat_controller.run_state_for(sid)
            state=app.persona_buddy_controller.snapshot().state
            if not states or states[-1]!=state: states.append(state)
            if run.status.value in {'completed','failed','blocked','stopped'}: break
            await asyncio.sleep(.03)
        await asyncio.sleep(.2)
        messages=store.messages_for_session(sid)
        replies=[m.content for m in messages if m.role is ConsoleMessageRole.ASSISTANT]
        result.update(run_status=run.status.value,visible_copy=run.visible_copy,buddy_states=states,final_buddy_state=app.persona_buddy_controller.snapshot().state,expected_reply=any('Migu live check passed' in r for r in replies),elapsed_seconds=round(time.monotonic()-start,2),app_exception=type(app._exception).__name__ if app._exception else None)
        screen.action_new_console_tab()
        await pilot.pause(.3)
        sid=store.active_session_id
        store.set_session_project_instruction_state(sid,ProjectInstructionControlState.legacy_disabled())
        composer=screen.query_one('#console-native-composer',ConsoleComposerBar)
        composer.load_draft('Count from 1 to 2000, one number per line, without commentary.')
        await screen._send_console_message_from_visible_action()
        stop_start=time.monotonic()
        while time.monotonic()-stop_start < 40:
            stop_run=app.console_runtime.chat_controller.run_state_for(sid)
            streamed_text=sum(len(m.content) for m in store.messages_for_session(sid) if m.role is ConsoleMessageRole.ASSISTANT)
            if stop_run.status.value == 'streaming':
                break
            if stop_run.status.value in {'completed','failed','blocked','stopped'}:
                break
            await asyncio.sleep(.01)
        result['stop_gate']='status-only; no first-token wait'
        result['stop_requested_from_status']=stop_run.status.value
        result['assistant_characters_before_stop']=streamed_text
        result['stop_exercised_during_stream']=stop_run.status.value == 'streaming'
        if result['stop_exercised_during_stream']:
            action_start=time.monotonic()
            await screen._stop_console_generation_from_visible_action()
            while time.monotonic()-stop_start < 45:
                stop_run=app.console_runtime.chat_controller.run_state_for(sid)
                if stop_run.status.value in {'completed','failed','blocked','stopped'}:
                    break
                await asyncio.sleep(.01)
            await pilot.pause(.2)
        stop_run=app.console_runtime.chat_controller.run_state_for(sid)
        result['stop_final_status']=stop_run.status.value
        result['stop_visible_copy']=stop_run.visible_copy
        result['stopped_by_user_marker']=any(m.role is ConsoleMessageRole.SYSTEM and m.content == 'Response stopped by user.' for m in store.messages_for_session(sid))
        result['stop_final_buddy_state']=app.persona_buddy_controller.snapshot().state
        result['stop_elapsed_seconds']=round(time.monotonic()-stop_start,2)
        result['stop_action_to_terminal_seconds']=round(time.monotonic()-action_start,3) if result['stop_exercised_during_stream'] else None
        result['stop_assistant_message_ids']=[m.id for m in store.messages_for_session(sid) if m.role is ConsoleMessageRole.ASSISTANT]
    result['app_exception']=type(app._exception).__name__ if app._exception else None
    result['normal_config_unchanged']=hashlib.sha256(NORMAL.read_bytes()).hexdigest()==normal_digest
    (OUT/'provider.json').write_text(json.dumps(result,indent=2)+'\n')
    print(json.dumps(result),flush=True)

if __name__=='__main__':
    if '--provider' in sys.argv:
        asyncio.run(provider())
    elif '--native' in sys.argv:
        app = TldwCli()
        def capture_geometry():
            try:
                view=app.screen._persona_buddy_view
                region=view.region
                (OUT/'native-geometry.json').write_text(json.dumps({'region':dict(x=region.x,y=region.y,width=region.width,height=region.height),'screen':type(app.screen).__name__,'pid':os.getpid(),'import_root':str(Path(sys.modules['tldw_chatbook'].__file__).parent)}))
            except (AttributeError, RuntimeError):
                pass
        app.call_after_refresh(lambda: app.set_interval(.2,capture_geometry))
        app.run()
    else:
        asyncio.run(speech())
